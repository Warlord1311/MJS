class A2_Q8_SimpleCalculator {
 public static double calculate(double a, double b, char op) {
     if (op == '+') return a + b;
     else if (op == '-') return a - b;
     else if (op == '*') return a * b;
     else if (op == '/') {
         if (b == 0) {
             System.out.println("Error: Division by zero!");
             return 0;
         }
         return a / b;
     } else {
         System.out.println("Invalid operator!");
         return 0;
     }
 }

 public static void main(String[] args) {
     Scanner sc = new Scanner(System.in);
     System.out.print("Enter first number, operator (+, -, *, /), and second number: ");
     double n1 = sc.nextDouble();
     char op = sc.next().charAt(0);
     double n2 = sc.nextDouble();
     System.out.println("Result: " + calculate(n1, n2, op));
 }
}

class A3_Q6_FindDuplicates {
 public static void main(String[] args) {
     Scanner sc = new Scanner(System.in);
     System.out.print("Enter number of elements (n): ");
     int n = sc.nextInt();
     int[] arr = new int[n];
     for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
     System.out.println("Duplicate elements:");
     for (int i = 0; i < n; i++) {
         boolean printed = false;
         for (int k = 0; k < i; k++) {
             if (arr[i] == arr[k]) {
                 printed = true;
                 break;
             }
         }
         if (printed) continue;
         for (int j = i + 1; j < n; j++) {
             if (arr[i] == arr[j]) {
                 System.out.println(arr[i]);
                 break;
             }
         }
     }
 }
}

interface DiscountApplicable {

}

abstract class Booking {
 String bookingId;
 String passengerName;
 String source;
 String destination;
 int numberOfPassengers;
 static int totalBookings = 0;

 public Booking(String bookingId, String passengerName, String source, String destination, int numberOfPassengers) {
     this.bookingId = bookingId;
     this.passengerName = passengerName;
     this.source = source;
     this.destination = destination;
     this.numberOfPassengers = numberOfPassengers;
     totalBookings++;
 }

 public void displayBookingDetails() {
     System.out.println("Booking ID           : " + bookingId);
     System.out.println("Passenger Name       : " + passengerName);
     System.out.println("Source               : " + source);
     System.out.println("Destination          : " + destination);
     System.out.println("Number of Passengers : " + numberOfPassengers);
 }

 public abstract double calculateFare();
}

class BusBooking extends Booking implements DiscountApplicable {
 String busNumber;
 String busType;
 String seatNumber;
 double farePerPassenger;

 public BusBooking(String bookingId, String passengerName, String source, String destination,
                   int numberOfPassengers, String busNumber, String busType, String seatNumber, double farePerPassenger) {
     super(bookingId, passengerName, source, destination, numberOfPassengers);
     this.busNumber = busNumber;
     this.busType = busType;
     this.seatNumber = seatNumber;
     this.farePerPassenger = farePerPassenger;
 }

 @Override
 public double calculateFare() {
     return farePerPassenger * numberOfPassengers;
 }

 @Override
 public double calculateDiscount(double fare) {
     if (numberOfPassengers >= 4) {
         return fare * 0.10; // 10% discount for 4 or more passengers
     }
     return 0.0;
 }

 public void displayCompleteDetails() {
     System.out.println("\n----- BUS BOOKING DETAILS -----");
     super.displayBookingDetails();
     System.out.println("Bus Number           : " + busNumber);
     System.out.println("Bus Type             : " + busType);
     System.out.println("Seat Number          : " + seatNumber);
     System.out.println("Fare Per Passenger   : Rs. " + farePerPassenger);
     double totalFare = calculateFare();
     double discount = calculateDiscount(totalFare);
     double finalAmount = totalFare - discount;
     System.out.println("Total Fare           : Rs. " + totalFare);
     System.out.println("Discount             : Rs. " + discount);
     System.out.println("Final Payable Amount : Rs. " + finalAmount);
 }
}

class CourierManagement {

 @SuppressWarnings("unchecked")
 private static ArrayList<CourierPackage> readPackagesFromFile() {
     File file = new File(FILE_NAME);
     if (!file.exists()) return new ArrayList<>();
     try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
         return (ArrayList<CourierPackage>) ois.readObject();
     } catch (IOException | ClassNotFoundException e) {
         return new ArrayList<>();
     }
 }

 private static void writePackagesToFile(ArrayList<CourierPackage> list) {
     try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(FILE_NAME))) {
         oos.writeObject(list);
     } catch (IOException e) {
         System.out.println("Error writing to file: " + e.getMessage());
     }
 }

abstract class TravelPackage {
 String packageId;
 String customerName;
 String destination;
 int numberOfDays;
 int numberOfPersons;
 static int totalPackages = 0;

 public TravelPackage(String packageId, String customerName, String destination, int numberOfDays, int numberOfPersons) {
     this.packageId = packageId;
     this.customerName = customerName;
     this.destination = destination;
     this.numberOfDays = numberOfDays;
     this.numberOfPersons = numberOfPersons;
     totalPackages++;
 }

 public void displayPackageDetails() {
     System.out.println("Package ID        : " + packageId);
     System.out.println("Customer Name     : " + customerName);
     System.out.println("Destination       : " + destination);
     System.out.println("Number of Days    : " + numberOfDays);
     System.out.println("Number of Persons : " + numberOfPersons);
 }

 public abstract double calculateCost();
}

class DomesticPackage extends TravelPackage implements DiscountApplicable {
 String hotelName;
 String transportMode;
 String mealPlan;
 double costPerPerson;

 public DomesticPackage(String packageId, String customerName, String destination, int numberOfDays,
                        int numberOfPersons, String hotelName, String transportMode, String mealPlan, double costPerPerson) {
     super(packageId, customerName, destination, numberOfDays, numberOfPersons);
     this.hotelName = hotelName;
     this.transportMode = transportMode;
     this.mealPlan = mealPlan;
     this.costPerPerson = costPerPerson;
 }

 @Override
 public double calculateCost() {
     return costPerPerson * numberOfPersons;
 }

 @Override
 public double calculateDiscount(double cost) {
     if (numberOfPersons >= 4) {
         return cost * 0.10; // 10% discount if 4 or more persons
     }
     return 0.0;
 }

 public void displayFullPackage() {
     System.out.println("\n----- DOMESTIC PACKAGE DETAILS -----");
     super.displayPackageDetails();
     System.out.println("Hotel Name        : " + hotelName);
     System.out.println("Transport Mode    : " + transportMode);
     System.out.println("Meal Plan         : " + mealPlan);
     System.out.println("Cost Per Person   : Rs. " + costPerPerson);
     double totalCost = calculateCost();
     double discount = calculateDiscount(totalCost);
     double finalPayable = totalCost - discount;
     System.out.println("Total Cost        : Rs. " + totalCost);
     System.out.println("Discount          : Rs. " + discount);
     System.out.println("Final Payable Amt : Rs. " + finalPayable);
 }
}

class MovieTicketManagement {
 private static final String FILE_NAME = "movie_bookings.dat";
 private static Scanner sc = new Scanner(System.in);

 @SuppressWarnings("unchecked")
 private static ArrayList<MovieTicket> readBookingsFromFile() {
     File file = new File(FILE_NAME);
     if (!file.exists()) return new ArrayList<>();
     try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
         return (ArrayList<MovieTicket>) ois.readObject();
     } catch (IOException | ClassNotFoundException e) {
         return new ArrayList<>();
     }
 }

 private static void writeBookingsToFile(ArrayList<MovieTicket> list) {

 public static void generateBookingReport() {

         System.out.println("No booking records available for report.");
         return;
     }
     int confirmedCount = 0;
     double confirmedRevenue = 0.0;
     MovieTicket highestTicketBooking = list.get(0);

     for (MovieTicket mt : list) {
         if (mt.bookingStatus.equalsIgnoreCase("Confirmed")) {
             confirmedCount++;
             confirmedRevenue += (mt.numberOfTickets * mt.ticketPrice);
         }
         if (mt.numberOfTickets > highestTicketBooking.numberOfTickets) {
             highestTicketBooking = mt;
         }
     }

     System.out.println("\n===== MOVIE BOOKING REPORT =====");
     System.out.println("Confirmed Bookings     : " + confirmedCount);
     System.out.println("Total Booking Revenue  : Rs. " + confirmedRevenue);
     System.out.println("Highest Ticket Booking : " + highestTicketBooking.bookingId);
     System.out.println("Movie Name             : " + highestTicketBooking.movieName);
     System.out.println("Number of Tickets      : " + highestTicketBooking.numberOfTickets);
 }

 public static void initializeDatabase() {

          Statement stmt = conn.createStatement()) {
         stmt.executeUpdate("CREATE DATABASE IF NOT EXISTS hotel_reservation");
         stmt.executeUpdate("USE hotel_reservation");
         stmt.executeUpdate("CREATE TABLE IF NOT EXISTS reservation ("
                 + "reservation_id INT PRIMARY KEY, "
                 + "guest_name VARCHAR(50), "
                 + "room_number VARCHAR(10), "

 public static void addReservation() {
     try (Connection conn = DriverManager.getConnection(DB_URL, USER, PASS)) {
         System.out.print("Enter Reservation ID: ");
         int id = Integer.parseInt(sc.nextLine().trim());
         System.out.print("Enter Guest Name: ");
         String guest = sc.nextLine().trim();
         System.out.print("Enter Room Number (e.g., R101): ");
         String roomNo = sc.nextLine().trim();

         // Check that room_number is not already registered for an active reservation (Reserved / Checked-In)
         String checkSql = "SELECT COUNT(*) FROM reservation WHERE room_number = ? AND status IN ('Reserved', 'Checked-In')";
         try (PreparedStatement checkStmt = conn.prepareStatement(checkSql)) {
             checkStmt.setString(1, roomNo);
             ResultSet rs = checkStmt.executeQuery();
             if (rs.next() && rs.getInt(1) > 0) {
                 System.out.println("Error: Room " + roomNo + " already has an active reservation!");
                 return;
             }
         }

         System.out.print("Enter Room Type (Standard / Deluxe / Suite): ");
         String type = sc.nextLine().trim();
         System.out.print("Enter Number of Stay Days: ");
         int days = Integer.parseInt(sc.nextLine().trim());
         System.out.print("Enter Room Rate per Day: ");
         double rate = Double.parseDouble(sc.nextLine().trim());
         System.out.print("Enter Status (Reserved / Checked-In / Checked-Out): ");
         String status = sc.nextLine().trim();

         String insertSql = "INSERT INTO reservation VALUES (?, ?, ?, ?, ?, ?, ?)";
         try (PreparedStatement pstmt = conn.prepareStatement(insertSql)) {
             pstmt.setInt(1, id);
             pstmt.setString(2, guest);
             pstmt.setString(3, roomNo);
             pstmt.setString(4, type);
             pstmt.setInt(5, days);
             pstmt.setDouble(6, rate);
             pstmt.setString(7, status);
             pstmt.executeUpdate();
             System.out.println("Reservation added successfully!");
         }
     } catch (SQLException e) {
         System.out.println("Insert Error: " + e.getMessage());
     }
 }
